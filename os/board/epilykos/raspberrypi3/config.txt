# Raspberry Pi firmware config.txt — Pi 3B (Tier 1).
# The Pi's own bootloader (start.elf) reads this before U-Boot runs; it
# hands off to u-boot.bin, which runs boot.scr (see boot.cmd).

kernel=u-boot.bin
arm_64bit=1
enable_uart=1
uart_2ndstage=1
disable_overscan=1

# The header UART (GPIO 14/15) stays on the mini-UART (ttyS1 under the
# mainline device tree) so the PL011 stays with the onboard Bluetooth
# chip (C-RUNTIME-003). enable_uart=1 pins the core clock so the
# mini-UART's baud rate doesn't drift; that is what makes the serial
# console usable for C-BOOT-000 evidence.
